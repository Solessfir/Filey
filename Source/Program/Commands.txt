// command Name {
//     label   = "Shown Label^search,aliases"   palette/menu label; omit for internal commands
//     tooltip = "One sentence."
//     hotkey  = "Ctrl+Shift+Key"               repeatable; Key is a Key_ name without the prefix; "G G" is G pressed twice

// }
//
// Hotkeys match modifiers exactly, so every accepted combination is listed.
// A character key without modifiers but Shift, such as "J" or "Shift+G", runs its command only while the Character Key Action option says so.
